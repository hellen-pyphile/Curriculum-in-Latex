\entry{Genpsiq -- Divulgação e Apoio Acadêmico em Genética Psiquiátrica}{}
  {Trabalho de Conclusão de Curso Técnico (ETEC)}{2024}
\begin{itemize}
  \item \textbf{Stack:} plataforma web com HTML5, CSS3 (interface responsiva),
        JavaScript (interatividade no cliente) e PHP (regras de negócio no servidor).
  \item \textbf{Pesquisa e design:} levantamento com 52 estudantes, dos quais 76,9\%
        relataram dificuldade para encontrar artigos científicos; prototipação
        iterativa no Canva para refinar a usabilidade.
  \item \textbf{Funcionalidades:} autenticação (login e cadastro); gerenciamento de
        arquivos (upload restrito a PDF e DOCX, indexação por ID, busca filtrada e
        exclusão); simulador de casos clínicos com validação automatizada de
        anamneses e relatórios psicológicos.
\end{itemize}
